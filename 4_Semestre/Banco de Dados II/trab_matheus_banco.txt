10. Crie uma nova tabela chamada vendas com os campos id, data_venda, valor, cliente_id.

CREATE TABLE vendas (
    id SERIAL PRIMARY KEY,
    data_venda DATE,
    valor NUMERIC(10,2),
    cliente_id INTEGER REFERENCES clientes(id)
);


11. Crie um índice composto para data_venda e valor em vendas.

CREATE INDEX idx_data_valor ON vendas(data_venda, valor);


12. Crie um índice baseado em expressão que indexe o nome em letras maiúsculas da tabela clientes.

CREATE INDEX idx_nome_upper ON clientes(UPPER(nome));


13. Crie uma view materializada com dados de vendas e crie um índice nela.

CREATE MATERIALIZED VIEW vendas_resumo AS
SELECT 
    cliente_id, 
    COUNT(*) AS total_vendas, 
    SUM(valor) AS total_valor
FROM vendas
GROUP BY cliente_id;

-- Índice na view materializada
CREATE INDEX idx_resumo_cliente ON vendas_resumo(cliente_id);
