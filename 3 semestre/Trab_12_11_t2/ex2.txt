
DROP TABLE IF EXISTS relacao_alocacao_funcao;
DROP TABLE IF EXISTS alocacao;
DROP TABLE IF EXISTS servidor;
DROP TABLE IF EXISTS vinculo;
DROP TABLE IF EXISTS funcao;
DROP TABLE IF EXISTS cargo;
DROP TABLE IF EXISTS escolaridade;
DROP TABLE IF EXISTS departamento;

CREATE TABLE escolaridade (
    id_escolaridade SERIAL PRIMARY KEY,
    descricao VARCHAR(80)
);


CREATE TABLE cargo (
    id_cargo SERIAL PRIMARY KEY,
    nome VARCHAR(80),
    salario DOUBLE PRECISION,
    id_escolaridade INTEGER,
    FOREIGN KEY (id_escolaridade) REFERENCES escolaridade(id_escolaridade)
);


CREATE TABLE funcao (
    id_funcao SERIAL PRIMARY KEY,
    nome VARCHAR(80),
    descricao VARCHAR(200),
    id_escolaridade INTEGER,
    FOREIGN KEY (id_escolaridade) REFERENCES escolaridade(id_escolaridade)
);


CREATE TABLE departamento (
    id_departamento SERIAL PRIMARY KEY,
    nome VARCHAR(80),
    descricao VARCHAR(200)
);


CREATE TABLE vinculo (
    id_vinculo SERIAL PRIMARY KEY,
    obs VARCHAR(200),
    descricao VARCHAR(200)
);


CREATE TABLE servidor (
    id_funcionario SERIAL PRIMARY KEY,
    nome VARCHAR(80),
    matricula VARCHAR(15),
    data_nascimento DATE,
    data_admissao DATE,
    endereco VARCHAR(80),
    bairro VARCHAR(80),
    cidade VARCHAR(15),
    cep VARCHAR(15),
    uf CHAR(2),
    telefone VARCHAR(15),
    email VARCHAR(80),
    senha VARCHAR(15),
    id_cargo INTEGER,
    id_vinculo INTEGER,
    FOREIGN KEY (id_cargo) REFERENCES cargo(id_cargo),
    FOREIGN KEY (id_vinculo) REFERENCES vinculo(id_vinculo)
);


CREATE TABLE alocacao (
    id_alocacao SERIAL PRIMARY KEY,
    data_saida DATE,
    data_inicio DATE,
    id_departamento INTEGER,
    id_funcionario INTEGER,
    FOREIGN KEY (id_departamento) REFERENCES departamento(id_departamento),
    FOREIGN KEY (id_funcionario) REFERENCES servidor(id_funcionario)
);


CREATE TABLE relacao_alocacao_funcao (
    id_alocacao SERIAL,
    id_funcao INTEGER,
    id_rel_aloc INTEGER PRIMARY KEY,
    FOREIGN KEY (id_alocacao) REFERENCES alocacao(id_alocacao),
    FOREIGN KEY (id_funcao) REFERENCES funcao(id_funcao)
);
